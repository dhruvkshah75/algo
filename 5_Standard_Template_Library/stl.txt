NOTES:
**************** PAIRS *************
    pair<int, string> p;   --------- can make pair of any two data types
    p = make_pair(2, "Dhruv")
    or 
    p = {2, "Dhruv"}
    to print the values:
    cout << p.first << p.second;
    
    pairs can directly be assigned 
    pair<int, string> p1 = p  --> this is done by passing it by value (not the memory address)
    pair<int, string> &p1 = p ---> this is done by passing it by reference (c++ references)
    p1.first = 3 so the value of it changes in its address



********************************** Iterators **********************************
vector<int>::iterator it;
it = v.begin(); or it = v.end();

    to directly print the values of elements of the vector without using iterators

    vector<int> v;
    for(int value: v){
        cout << value << " ";
    }
    vector<pair<int,int>> v1;
    for(pair<int,int> &value: v1){   // pass by reference
        cout << value.first << " " << value.second << endl;
    }
